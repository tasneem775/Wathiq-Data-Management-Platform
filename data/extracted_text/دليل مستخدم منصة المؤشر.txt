 تدعـم منصـة قيـاس المؤشـر الوطنـي للبيانـات “نضـىء” قيـاس نضـج إدارة البيانـات فـي الجهـات الحكوميـة
 وقيــاس االمتثــال لضوابــط ومواصفــات إدارة البيانــات الوطنيــة وقيــاس مســتوى تقــدم الجهــة مــن حيــث
 ،االسـتفادة مـن المنصـات الوطنيـة وذلـك مـن خالل تقييـم العمليـات التشـغيلية إلدارة البيانـات لـدى الجهـة
 حيـث سـتوفر منصـة المؤشـر الوطنـي للبيانـات نتيجـة نهائيـة لقيـاس النضـج واالمتثـال وقيـاس المؤشـرات
التــشغيلية ــفي الجــهات الحكومــية وذــلك لدعمــها ــفي تحــسين ممارــسات إدارة البياــنات الوطنــية
 ويأتـي هـذا القيـاس بنـاًءً علـى وثيقـة المؤشـر الوطنـي للبيانـات، لدعـم الجهـات علـى فهـم الفجـوات وتحديـد
مــجاالت التحــسين وتقدــيم التوصــيات لرــفع مــستوى نــضج الجــهة ــفي ممارــسات إدارة البياــنات
 تتحمــل كل جهــة حكوميــة مســؤولية صحــة البيانــات واألدلــة المرفقــة، حيــث أن قيــاس النضــج واالمتثــال
 سـيعتمد علـى األدلـة الداعمـة المرفقـة مـن قبـل الجهـة، ولـن يتـم قبـول أي مرفـق بعـد انتهـاء الفتـرة الزمنيـة
لتعبــئة االــستبيان وــسُتُ عامل كاــفة المعلوــمات والمرفــقات بــسرية تاــمة وــلن يــتم نــشر أي ــجزء منــها
 ،تعتبـر الجهـة مسـؤولة عـن الحفـاظ علـى سـرية وتقييـد الوصـول إلـى بيانـات المنصـة وتفويـض الصالحيـات
 وتفاصيـل معلومـات الحسـاب، وتتحمـل المسـؤولية التامـة عـن جميـع األنشـطة التـي تحـدث فيمـا يتعلـق
